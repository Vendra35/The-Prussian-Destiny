# --- THE RECURRING CHOICE, every two years while the crisis runs ---
# Both answers are real. Conceding ends it with the crown weakened and the
# realm intact; repression ends it with the realm cowed and the treasury spent.
pd_bohemia.10 = {
	type = country_event
	title = pd_bohemia.10.title
	desc = pd_bohemia.10.desc
	option = {
		name = pd_bohemia.10.a
		change_variable = {
			name = PD_boh_concessions
			add = 25
		}
		# The crown yields ground to keep the realm together.
		add_legitimacy = -4
		ai_chance = {
			factor = 50
		}
	}
	option = {
		name = pd_bohemia.10.b
		change_variable = {
			name = PD_boh_repression
			add = 25
		}
		# Soldiers and confiscations are not free.
		add_gold = -50
		add_prestige = -2
		ai_chance = {
			factor = 50
		}
	}
}
