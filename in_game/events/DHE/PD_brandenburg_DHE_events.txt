# Brandenburg için 1512 veya 1618 yılları arasında 2. situation başlamadan prusya formlama DHE eventi.
# NOT: Vanilla form_country yolu kasıtlı olarak kapatılmadı, ama pratikte
# risksiz — PRU_f, "Military Order" reformuna sahip olmamayı şart koşuyor
# ve TEU bu reformu ancak DHE zincirinin sekülerleşme sürecinden geçerek
# kaybediyor. O yüzden situation açıkken üçüncü bir gücün (ya da TEU'nun
# reform hâlâ üstündeyken) PRU_f'i vanilla yoldan formlaması mümkün değil.
pd_brandenburg_dhe.4 = {
	type = country_event
	title = pd_brandenburg_dhe.4.title
	desc = pd_brandenburg_dhe.4.desc
	#image = "gfx/interface/illustrations/situation/placeholder.dds"
	fire_only_once = yes
	dynamic_historical_event = {
		tag = BRA
		from = 1490.1.1
		to = 1700.1.1
		monthly_chance = 100
	}
	trigger = {
		tag = BRA
		at_war = no
		# Softlock Failsafe: Ya vassalın olacak, ya da tamamen yutmuş olacaksın.
		OR = {
			# Orijinal İhtimal: Tötonlar yaşıyor ve vassalım
			AND = {
				country_exists = c:TEU
				c:TEU = {
					is_subject_of = ROOT
				}
			}
			# Fallback İhtimal: Tötonlar yok ve toprakları bende
			AND = {
				OR = {
					# Töton tamamen haritadan silindiyse
					NOT = {
						country_exists = c:TEU
					}
					# VEYA Prusya, Brandenburg ve Pomeranya'nın HİÇBİRİNDE toprağı kalmadıysa
					NOT = {
						c:TEU = {
							OR = {
								has_presence_in = area:prussia_area
								has_presence_in = area:brandenburg_area
								has_presence_in = area:pomerania_area
							}
						}
					}
				}
				c:BRA = {
					OR = {
						# Ya Danzig Ya da Königsberg Brandenburg'un elinde olmalı!
						owns = location:konigsberg
						owns = location:gdansk
					}
				}
			}
		}
		# Dinamik Tarih Kontrolü
		OR = {
			# Frontloaded: 1512 sonrası tetiklenir
			AND = {
				has_game_rule = PD_timeline_frontloaded
				current_date > 1512.1.1
			}
			# Strict Historical: 1618 sonrası tetiklenir
			AND = {
				has_game_rule = PD_timeline_strict_historical
				current_date > 1618.1.1
			}
		}
	}
	immediate = {
	}
	option = {
		name = pd_brandenburg_dhe.4.a
		# Hata önleyici: Sadece Tötonlar hala yaşıyorsa VE o 3 bölgeden birinde varlığı varsa annexle
		# 1) TEU hâlâ yaşıyor ve ROOT'un vassalıysa → mevcut annex bloğu aynen kalıyor
		if = {
			limit = {
				country_exists = c:TEU
				c:TEU = {
					is_subject_of = ROOT
					OR = {
						has_presence_in = area:prussia_area
						has_presence_in = area:brandenburg_area
						has_presence_in = area:pomerania_area
					}
				}
			}
			ROOT = {
				annex_country = {
					country = c:TEU
					reason = Diplomatic
					transfer = yes
				}
			}
		}
		# 2) YENİ: PRU zaten varsa VE birinin (kim olursa olsun) vassalıysa
		# → tepedeki gerçek overlord'u bul, o kendi vassalını yutsun.
		# Mongol modundaki idiomla aynı: top_overlord_or_this, overlordu yoksa
		# kendisini döndürür — o yüzden "bağımsız mı" kontrolünü buradan yapıyoruz.
		if = {
			limit = {
				country_exists = c:PRU
				NOT = {
					c:PRU = {
						# bağımsızsa top_overlord = kendisi
						top_overlord_or_this ?= c:PRU
					}
				}
			}
			c:PRU = {
				top_overlord_or_this = {
					save_temporary_scope_as = pru_true_overlord
				}
			}
			scope:pru_true_overlord = {
				annex_country = {
					country = c:PRU
					reason = Diplomatic
					transfer = yes
				}
			}
		}
		# 3) Artık PRU yok (hiç yoktu, ya da az önce overlord'u tarafından yutuldu) → forma.
		if = {
			limit = {
				NOT = {
					country_exists = c:PRU
				}
			}
			form_country = formable_country:PRU_f
		}
	}
}
