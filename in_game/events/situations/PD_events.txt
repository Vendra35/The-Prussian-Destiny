pd_brandenburg.108 = {
	type = country_event
	category = situation_event
	title = pd_brandenburg.108.title
	historical_info = pd_brandenburg.108.historical_info
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					country_exists = scope:rival_country
				}
				desc = pd_brandenburg.108.desc.1
			}
			triggered_desc = {
				desc = pd_brandenburg.108.desc.2
			}
		}
	}
	#image = "gfx/interface/illustrations/situation/placeholder.dds"
	immediate = {
		situation:the_prussian_ambition = {
			var:PD_conquest_target_country = {
				save_scope_as = target_country
			}
		}
		scope:target_country = {
			ordered_country = {
				limit = {
					OR = {
						is_enemy_of = prev
						is_rival_of = prev
					}
					this != root
					this.capital.continent = continent:europe
					NOT = {
						any_owned_location = {
							OR = {
								this.area = area:brandenburg_area
								this.area = area:pomerania_area
								this.area = area:prussia_area
							}
						}
					}
				}
				max = 1
				order_by = expected_army_size
				save_scope_as = rival_country
			}
		}
	}
	# Option 1
	option = {
		trigger = {
			country_exists = scope:rival_country
		}
		### Rivaldan yard?m iste!
		name = pd_brandenburg.108.a
		# The coalition is what actually broke the Order: in 1454 its own Prussian Confederation renounced allegiance and offered Prussia to Poland.
		historical_option = yes
		scope:rival_country = {
			trigger_event_non_silently = pd_brandenburg.109
		}
		ai_chance = {
			factor = 1000
			modifier = {
				factor = 0
				scope:target_country = {
					num_locations <= 15
				}
			}
		}
	}
	#Option 2
	option = {
		trigger = {
			country_exists = scope:rival_country
		}
		name = pd_brandenburg.108.b
		add_prestige = 10
		ai_chance = {
			factor = 1
		}
		situation:the_prussian_ambition = {
			set_variable = {
				name = PD_conquest_target_rival
				value = 1
			}
		}
	}
	# Option 3 - Vay G?t.
	option = {
		trigger = {
			NOT = {
				country_exists = scope:rival_country
			}
		}
		name = pd_brandenburg.108.c
		situation:the_prussian_ambition = {
			set_variable = {
				name = PD_conquest_target_rival
				### Can't get another rival this cycle
				value = 1
			}
		}
	}
}
