			}
			# 2. İHTİMAL: HRE birleşip tek ülke (Tag) olduysa, Ültimatomu direkt (HRE)ye yolla!
			else_if = {
				limit = {
					exists = c:HRE
				}
				c:HRE = {
					trigger_event_non_silently = pd_brandenburg.214
				}
			}
		}
		# 1. HEDEF SEÇME BLOĞU
		if = {
			limit = {
				var:PD_phase2_conquest_target = 0
				c:PRU = {
					at_war = no
				}
			}
			c:PRU = {
				ordered_neighbor_country = {
					limit = {
						is_rebel_country = no
						NOT = {
							has_truce_with = prev
						}
						NOT = {
							is_subject_of = prev
						}
						# The whole target test lives in one place -
						# PD_ascension_valid_target in common/scripted_triggers/ - because
						# the FALLBACK below has to apply exactly the same test to the
						# target it is already holding. Two copies drift: Mongol
						# Resurgence ended up with comments saying 1.1 over code saying
						# 0.7 for precisely that reason.
						PD_ascension_valid_target = yes
					}
					max = 1
					# Smallest first among equals - the easy war before the hard one,
					# and the gate above has already thrown out anything unwinnable.
					# But size must NOT outrank the objective: step 1 closes on Silesia
					# and nothing else, and Silesia's holder is usually the biggest
					# country on the list. Without the bonus below it sorts last, behind
					# every homeland squatter and - once diverted - every small western
					# state, so the one country that can close the step never gets
					# picked. Conditional order_by per country_interactions/hre.txt:2143.
					order_by = {
						value = num_locations
						multiply = -1
						if = {
							limit = {
								situation:the_prussian_ascension = {
									var:PD_phase2_current_step = 1
								}
								any_owned_location = {
									this.area = area:silesia_area
								}
							}
							add = 100000
						}
					}
					situation:the_prussian_ascension = {
						set_variable = {
							name = PD_phase2_conquest_target_country
							value = prev
						}
						set_variable = {
							name = PD_phase2_conquest_target
							value = 1
						}
					}
				}
			}
		}
		# SAYAÇ ARTIRMA
		change_variable = {
			name = PD_phase2_conquest_cooldown
			add = 1
		}
		# STEP STALL - months spent on the CURRENT step. Reset only when the
		# step changes, so it measures the thing that is actually stuck: a
		# war elsewhere does not clear it. The target gate reads it to open
		# its second tier at 120, and the switch below reads it at 240.
		change_variable = {
			name = PD_phase2_step_stall
			add = 1
		}
		# Twenty years on step 1 and Silesia is still out of reach. Open the
		# west, and record it, because the panel has to be able to say so.
		if = {
			limit = {
				var:PD_phase2_current_step = 1
				PD_ascension_stall_divert = yes
				var:PD_phase2_diverted = 0
			}
			set_variable = {
				name = PD_phase2_diverted
				value = 1
			}
		}
		# ...and close it again the moment Silesia is reachable. Without this
		# the diversion is permanent: the flag clears only on a step change,
		# and step 1 changes only when Silesia falls. So an enemy that grew
		# weak later would never be turned back for - Prussia would keep
		# marching west past an open door. The panel follows the flag, so it
		# stops saying 'marching west' on its own.
		if = {
			limit = {
				var:PD_phase2_current_step = 1
				var:PD_phase2_diverted = 1
				c:PRU = {
					any_neighbor_country = {
						any_owned_location = {
							this.area = area:silesia_area
						}
						PD_ascension_valid_target = yes
					}
				}
			}
			set_variable = {
				name = PD_phase2_diverted
				value = 0
			}
			# Back to 120, not 0: keep the tier-2 tolerance Prussia earned over
			# the last twenty years, but make it wait another decade before it
			# may divert again. At 0 the flag would flap month to month around
			# the gate boundary and the panel line would blink with it.
			if = {
				limit = {
					has_game_rule = BRA_buff_enabled
				}
				set_variable = {
					name = PD_phase2_step_stall
					value = 36
				}
			}
			else = {
				set_variable = {
					name = PD_phase2_step_stall
					value = 120
				}
			}
		}
		# --- GÜVENLİK SİGORTASI (FALLBACK) BRANDENBURG İÇİN ---
		if = {
			limit = {
				var:PD_phase2_conquest_target = 1
				OR = {
					# 1. Hedef ülke haritadan silindiyse (ilhak edildiyse) veya multiplayer'da scope koptuysa
					NOT = {
						country_exists = var:PD_phase2_conquest_target_country
					}
					# 2. Hedef bir şekilde bizim vasalımız olduysa veya araya ateşkes girdiyse
					var:PD_phase2_conquest_target_country = {
						OR = {
							is_subject_of = c:PRU
							has_truce_with = c:PRU
							# 3. The target no longer passes the test that picked it.
							# Selection and declaration are 12 to 60 months apart, which
							# is time enough for the target to gain an ally or for the
							# step to move on. Without this the war fires anyway, on a
							# flag set years earlier. Ported from Mongol Resurgence,
							# MR_mongol_resurgence.txt:534 (audited there 2026-08-28).
							NOT = {
								PD_ascension_valid_target = yes
							}
						}
					}
				}
			}
			# Hedef geçersiz! Sistemi sıfırla ki motor yeni hedef arasın:
			situation:the_prussian_ascension = {
				set_variable = {
					name = PD_phase2_conquest_target
					value = 0
				}
			}
		}
		# 2. SAVAŞ AÇMA BLOĞU
		if = {
			limit = {
				var:PD_phase2_conquest_target = 1
				OR = {
					# Terminator Mode: Hızlı agresif genişleme (1 yılda bir)
					AND = {
						has_game_rule = BRA_buff_enabled
						var:PD_phase2_conquest_cooldown > 12
					}
					# Historical / Vanilla Mode: Yavaş ve istikrarlı genişleme (5 yılda bir / 60 ay)
					AND = {
						NOT = {
							has_game_rule = BRA_buff_enabled
						}
						var:PD_phase2_conquest_cooldown > 60
					}
				}
			}
			c:PRU = {
				if = {
					limit = {
						OR = {
							# Terminator Mode: Son savaştan 1 yıl (12 ay) geçtiyse saldırabilir.
							AND = {
								has_game_rule = BRA_buff_enabled
								months_since_war >= 12
							}
							# Historical / Vanilla Mode: Son savaştan en az 2 yıl (24 ay) geçmiş olmalı.
							AND = {
								NOT = {
									has_game_rule = BRA_buff_enabled
								}
								months_since_war >= 24
							}
						}
						situation:the_prussian_ascension = {
							var:PD_phase2_conquest_target_country = {
								NOT = {
									is_subject_of = prev
								}
								NOT = {
									has_truce_with = prev
								}
							}
						}
					}
					# SAVAŞ İLANI EVENTİ (Hedefe dalıyoruz)
					trigger_event_silently = pd_brandenburg.203
				}
			}
		}
	}
	on_ending = {
		c:PRU = {
			if = {
				limit = {
					is_ai = yes
					NOT = {
						has_global_variable = PD_hegemony_lost
					}
					NOT = {
						has_game_rule = PD_blood_and_iron_mechanic_disabled
					}
				}
				add_country_modifier = {
					modifier = PD_preparing_for_blood_and_iron
					days = -1
				}
			}
			remove_country_modifier = PD_prussia_preparing_for_ascension
			remove_country_modifier = PD_brandenburg_ascension_campaign
			remove_country_modifier = PD_brandenburg_ascension_campaign_historical
			remove_country_modifier = PD_brandenburg_military_supremacy
			remove_country_modifier = PD_brandenburg_military_supremacy_historical
			remove_country_modifier = PD_brandenburg_zollverein_ambition
			remove_country_modifier = PD_imperial_outcast_situation_2
			# Başarılı olma şartı (Silezya, vb. almış mı VE Avusturya Savaşından zaferle çıkmış mı?)
			if = {
				limit = {
					# YENİ EKLENEN SAVAŞ/ZAFER KONTROLLERİ:
					# --- 1. HATA BURADAYDI, ÇÖZÜLDÜ: Değişkeni Situation'a soruyoruz! ---
					situation:the_prussian_ascension = {
						var:PD_phase2_current_step = 3
					}
					at_war = no
					# ÇÖZÜM BURADA: Ya değişkeni aldık ya da HRE lideri kalmadı (Hükmen Galibiyet)!
					OR = {
						has_variable = PD_hegemony_victory
						AND = {
							# HRE Organizasyonu DAĞILMIŞ olmalı...
							NOT = {
								exists = international_organization:hre
								exists = international_organization:hre.leader_country
							}
							# ...VE tek bir Mega-Devlete (Tag) DÖNÜŞMEMİŞ olmalı!
							# (Yoksa Prusya kaybettiği halde havadan zafer kazanır)
							NOT = {
								exists = c:HRE
							}
						}
					}
					NOT = {
						has_variable = PD_submitted_to_emperor
					}
					# Eğer ültimatomda korkup geri adım attıysa BAŞARISIZ sayılır.
					# BÖLGE KONTROLLERİ (Aynı Kalıyor):
					AND = {
						NOT = {
							area:silesia_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:upper_saxony_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:lower_saxony_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:mecklenburg_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:holstein_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:westphalia_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:hesse_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:rhineland_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
											this = c:FRA
										}
									}
								}
							}
						}
						NOT = {
							area:brandenburg_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:pomerania_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU
										}
									}
								}
							}
						}
						NOT = {
							area:prussia_area = {
								any_ownable_location_in_area = {
									owner = {
										NOR = {
											this = c:PRU
											is_subject_of = c:PRU

									}
								}
							}

					limit = {
						NOT = {
							has_game_rule = BRA_buff_disabled
						}

					add_country_modifier = {
						modifier = PD_the_new_german_order
						years = -1
					}
				}
				# #######################################################################
				# --- BÜYÜK ZAFER: KUZEY ALMAN FEDERASYONU (NGF) DOĞUYOR ---
				# #######################################################################
				# 1. Zollverein ile bağladığımız tüm küçük devletleri tek çatı altında topluyoruz (Anschluss)
				# --- İLHAK (ANNEX) İŞLEMİ DÜZELTİLDİ ---
				# ÇÖZÜM BURADA: HRE hala ayaktaysa (Barışla yıkılmamışsa veya hükmen kazanılmışsa), acımadan burada YIK!
				if = {
					limit = {
						exists = international_organization:hre
					}
					destroy_international_organization = {
						target = international_organization:hre
					}
				}
				every_subject = {
					limit = {
						OR = {
							has_presence_in = region:north_german_region
						}
					}
					# Prusya'ya dönüp diyoruz ki: "Şu an baktığımız vasalı (this) ilhak et."
					c:PRU = {
						annex_country = {
							country = prev							# 'prev', yani bir önceki scope olan vasalı (every_subject) kasteder
							reason = Diplomatic
							transfer = yes
						}
					}
				}
				# Başardıysa Kendine ve Spectator Ülkelere Event yolluyor haber veriyoruz.
				every_country = {
					limit = {
						this != c:PRU
						OR = {
							has_presence_in = region:north_german_region
							has_presence_in = region:south_german_region
							has_presence_in = region:baltic_region
							has_presence_in = region:france_region
							has_presence_in = region:italy_region
						}
					}
					# Prussia win the situaton and Great War
					trigger_event_non_silently = pd_brandenburg.204
				}
				c:PRU = {
					# Prussia win the situaton and Great War and Changing Tag to NGC Event.
					trigger_event_non_silently = pd_brandenburg.205
				}
			}
			else = {
				# BAŞARISIZ BİTİŞ CEZALARI
				# Başarısızsa Kendine ve Spectator Ülkelere Event yolluyor haber veriyoruz.
				every_country = {
					limit = {
						this != c:PRU
						OR = {
							has_presence_in = region:north_german_region
							has_presence_in = region:south_german_region
							has_presence_in = region:baltic_region
							has_presence_in = region:france_region
							has_presence_in = region:italy_region
						}
					}
					# Prussia lose
					trigger_event_non_silently = pd_brandenburg.206
				}
				c:PRU = {
					# Prussia lose
					trigger_event_non_silently = pd_brandenburg.207
				}
			}
		}
		# Temizlik
		# 2. KURBANLARIN KORKU MODIFIERINI TEMİZLEME
		every_country = {
			limit = {
				has_country_modifier = PD_brandenburg_imperial_anxiety
			}
			remove_country_modifier = PD_brandenburg_imperial_anxiety
		}
		remove_variable = PD_austria_ultimatum_sent
		remove_variable = PD_phase2_conquest_cooldown
		remove_variable = PD_phase2_conquest_target
		remove_variable = PD_phase2_conquest_target_country
		remove_variable = PD_zollverein_asked
		remove_variable = PD_hegemony_victory
		remove_variable = PD_submitted_to_emperor
		remove_variable = PD_phase2_current_step
		remove_variable = PD_phase2_step_stall
		remove_variable = PD_phase2_diverted
		remove_global_variable = prussian_ascension_active
	}

		#custom_tooltip = PD_PRUSSIAN_TARGET_LOCATION_TT_2
	}

	map_color = {
		if = {
			limit = {
				OR = {
					# 1. Şart: Toprak Prusya'nın mı?
					# tag = X reads a literal name instead of resolving c:X, so a
					# tag that has left the map is simply false here rather than
					# throwing once per location per redraw.
					owner ?= {
						tag = PRU
					}
					# 2. Şart: HRE hala yaşıyorsa VE Toprak HRE Liderinin mi?
					AND = {
						exists = international_organization:hre
						exists = international_organization:hre.leader_country
						owner ?= international_organization:hre.leader_country
					}
				}
			}
			# Eğer yukarıdakilerden biriyse, o ülkenin kendi rengini haritaya bas
			value = this.owner.country_color
		}
		else = {
			value = define:NMapColors|DEFAULT_COLOR
		}
	}
	secondary_map_color = {
		if = {
			limit = {
				PD_owner_is_subject_of = { TAG = PRU }
			}
			value = blue
		}
		else_if = {
			limit = {
				OR = {
					this.area = area:silesia_area
					this.area = area:upper_saxony_area
					this.area = area:lower_saxony_area
					this.area = area:mecklenburg_area
					this.area = area:holstein_area
					this.area = area:westphalia_area
					this.area = area:hesse_area
					this.area = area:rhineland_area
					this.area = area:brandenburg_area
					this.area = area:pomerania_area
					this.area = area:prussia_area
				}
				NOT = {
					owner ?= {
						tag = PRU
					}
				}
			}
			value = yellow			# Hedefleri sarı göster
		}
	}
