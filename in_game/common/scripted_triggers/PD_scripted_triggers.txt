# --- 1. SITUATION: PRUSSIAN DESTINY (SITUATION BİTİRME KONDİSYONUNA BAKMAK İÇİN ÇAĞRILIR) ---
prussian_ambition_end_trigger = {
	OR = {
		# Knockout 1a: Brandenburg'un DOĞRUDAN zaferi
		custom_tooltip = {
			text = prussian_ambition_knockout_teu_tt
			OR = {
				c:TEU = {
					is_subject_of = c:BRA
				}
				AND = {
					country_exists = c:TEU
					NOT = {
						c:TEU = {
							OR = {
								has_presence_in = area:prussia_area
								has_presence_in = area:brandenburg_area
								has_presence_in = area:pomerania_area
							}
						}
					}
					c:BRA = {
						OR = {
							owns = location:konigsberg
							owns = location:gdansk
						}
					}
				}
			}
		}
		# Knockout 1b: TEU üçüncü bir ülke tarafından tamamen yok edildi
		custom_tooltip = {
			text = prussian_ambition_teu_destroyed_by_other_tt
			AND = {
				NOT = {
					country_exists = c:TEU
				}
				OR = {
					AND = {
						has_game_rule = PD_timeline_frontloaded
						current_date > 1500.1.1
					}
					AND = {
						has_game_rule = PD_timeline_strict_historical
						current_date > 1525.1.1
					}
				}
			}
		}
		# Knockout 2: Tötonların Zaferi
		custom_tooltip = {
			text = prussian_ambition_knockout_bra_tt
			OR = {
				c:BRA = {
					is_subject_of = c:TEU
				}
				NOT = {
					country_exists = c:BRA
				}
				NOT = {
					c:BRA = {
						OR = {
							has_presence_in = area:prussia_area
							has_presence_in = area:brandenburg_area
							has_presence_in = area:pomerania_area
						}
					}
				}
			}
		}
		# Süre Sonu Puanlaması
		custom_tooltip = {
			text = prussian_ambition_time_limit_tt
			AND = {
				country_exists = c:TEU
				country_exists = c:BRA
				c:BRA = {
					at_war = no
				}
				c:TEU = {
					at_war = no
				}
				OR = {
					AND = {
						has_game_rule = PD_timeline_frontloaded
						current_date > 1500.1.1
					}
					AND = {
						has_game_rule = PD_timeline_strict_historical
						current_date > 1525.1.1
					}
				}
			}
		}
	}
}

# --- 3. SITUATION: BLOOD AND IRON (SITUATION BİTİRME KONDİSYONUNA BAKMAK İÇİN ÇAĞRILIR) ---
blood_and_iron_end_trigger = {
	OR = {
		# 1. İHTİMAL: NGC Haritadan Silindi (Başarısızlık)
		custom_tooltip = {
			text = blood_and_iron_ngc_destroyed_tt
			NOT = {
				OR = {
					country_exists = c:PRU
					country_exists = c:NGC
				}
			}
		}
		# 2. İHTİMAL: Fransa Yok Edildi/Başka Ülke Tarafından Yutuldu (İptal)
		custom_tooltip = {
			text = blood_and_iron_fra_destroyed_tt
			NOT = {
				country_exists = c:FRA
			}
		}
		# 3. İHTİMAL: Almanya Kuruldu (Büyük Zafer!)
		custom_tooltip = {
			text = blood_and_iron_victory_tt
			has_global_variable = german_empire_formed
		}
		# 4. İHTİMAL: Fransa Prusya'yı Parçaladı (Shattered Ambition)
		custom_tooltip = {
			text = blood_and_iron_defeat_tt
			has_global_variable = blood_and_iron_lost
		}
		# 5. İHTİMAL: JENERİK BARIŞ / ZAMAN AŞIMI (YENİ EKLENEN ÇÖZÜM)
		custom_tooltip = {
			text = blood_and_iron_stalemate_tt
			# Ems Telgrafı çoktan patlamış olmalı (Yani Kan ve Demir aşamasına geçilmiş olmalı)
			has_global_variable = ems_dispatch_fired
			# Özel zafer veya yıkım antlaşmaları KULLANILMAMIŞ olmalı
			NOT = {
				has_global_variable = german_empire_formed
			}
			NOT = {
				has_global_variable = blood_and_iron_lost
			}
			# Savaş bitmiş olmalı (Fransa artık Prusya veya NGC ile savaşta değil)
			c:FRA = {
				NOT = {
					is_at_war_with = c:PRU
				}
				NOT = {
					is_at_war_with = c:NGC
				}
			}
			# Tansiyon 200'ü geçmiş olmalı!
			# (Ems 100'de patlıyor. Bu, savaş kararı verildikten sonra oyunda en az 8-9 yıl geçmiş demek.
			# Oyuncu Fransa Event 304'te "Savaşı Erteleyeceğim (5 Yıl)" seçeneğini seçerse, Situation'ın erken kapanmasını KESİNLİKLE engeller!)
			var:PD_franco_prussian_tension > 450
		}
	}
}

### Army Balance
PD_percent_of_army_balance = {
	OR = {
		AND = {
			expected_army_size <= 12
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.9
			}
		}
		AND = {
			expected_army_size > 12
			expected_army_size <= 30
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.75
			}
		}
		AND = {
			expected_army_size > 30
			expected_army_size <= 50
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.6
			}
		}
		AND = {
			expected_army_size > 50
			expected_army_size <= 120
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.45
			}
		}
		AND = {
			expected_army_size > 120
			regular_army_size < {
				value = root.expected_army_size
				multiply = 0.3
			}
		}
	}
}
