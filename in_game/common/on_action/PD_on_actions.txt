# 1. Situation Başlatıcı (Hem Oyuncu Hem AI İçin)

	trigger = {
		tag = BRA
	}
	events = {
		pd_brandenburg.10
	}
}

# 2. Situation Başlatıcı (Hem Oyuncu Hem AI İçin)
on_brandenburg_situation_2_pulse = {
	trigger = {
		tag = PRU
	}
	events = {
		pd_brandenburg.299
	}
}

# 2. AI Otomatik Bina Basımı
on_brandenburg_manpower_pulse = {
	trigger = {
		OR = {
			tag = BRA
			tag = PRU
		}
		is_ai = yes
	}
	events = {
		pd_brandenburg.999
	}
}

# 3. AI Otomatik Asker Basımı (Sadece barış zamanı)
on_brandenburg_construct_army_pulse = {
	trigger = {
		OR = {
			tag = BRA
			tag = PRU
			tag = NGC
		}
		is_ai = yes
		at_war = no
	}
	events = {
		pd_brandenburg.998
	}
}
