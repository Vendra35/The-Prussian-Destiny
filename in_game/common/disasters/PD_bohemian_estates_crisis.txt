# THE BOHEMIAN ESTATES CRISIS
#
# A disaster for whoever rules Bohemia. It is CONTENT, not a lever: nothing
# here names Prussia, the Ascension or Silesia, and there is no is_ai gate, so
# a player and the AI live under the same rules. If it gives Prussia an opening
# that is emergent - the Ascension's target gate is a strength ratio and this
# makes Bohemia weaker. Do not couple it to that situation later.
#
# Two historical anchors, one per timeline rule (see docs/BOHEMIA-DISASTER.md):
#   frontloaded      - the Bohemian Revolt, 1618-1620. Prague's governors go
#                      out of a window, the estates depose their king, and
#                      Ferdinand breaks them at the White Mountain.
#   strict historical - the invasion of 1741-42. A Franco-Bavarian army takes
#                      Prague and Charles Albert is crowned there, while Maria
#                      Theresa fights for her inheritance. That is the actual
#                      historical reason Silesia was takeable.
#
# The gate is LEGITIMACY, because ruler_religion is not a trigger and because
# both anchors are legitimacy crises by definition - a king deposed, and a
# rival crowned in the capital. It is also the fair condition: a Bohemia that
# keeps its crown secure never sees this at all.

PD_bohemian_estates_crisis = {

	image = "gfx/interface/illustrations/disaster/PD_bohemian_estates_crisis.dds"

	monthly_spawn_chance = monthly_spawn_chance_unique

	fire_only_once = yes

	can_start = {
		# By land, not by tag: BOH, HAB or whoever actually holds the crown.
		owns = location:prague
		has_ruler = yes
		# TWO ways in, and both are about the estates - which is what
		# this disaster is. A legitimacy or stability path was tried and
		# dropped: a country whose stability had merely dipped would have
		# got 'the estates against the crown' with no estates in it.
		OR = {
			# Set by pd_bohemia_dhe.1, the historical event that makes this
			# crisis reliable rather than hopeful. It is a flag, not a sum, so
			# no retuning of the thresholds below can quietly disarm it.
			custom_tooltip = {
				text = PD_boh_start_grievance_tt
				has_variable = PD_boh_estates_aggrieved
			}
			custom_tooltip = {
				text = PD_boh_start_nobles_tt
				country_has_estate = estate_type:nobles_estate
				estate_satisfaction:nobles_estate < 0.5
			}
		}
		# Vanilla's own guards, turmoil_in_brandenburg.txt:11-12.
		in_civil_war = no
		has_any_active_disaster = no
		# Shared with the historical event that guarantees this crisis - see
		# PD_bohemian_era_window. An earlier draft gated it at 1600 and 1725,
		# both ~78 years after their situation can start, so the crisis could
		# only ever have arrived once the story was over.
		PD_bohemian_era_window = yes
	}

	can_end = {
		PD_bohemian_estates_crisis_end_trigger = yes
	}

	# THE WEAKENING, and it has to be real.
	#
	# The first version of this block was bookkeeping - control, legitimacy,
	# a manpower trim. None of it moved anything the Ascension's target gate
	# reads, and that gate is the whole reason a weak Bohemia matters:
	# defensive_alliance_strength is "the nation WITH all countries giving
	# defensive support" (triggers.log:3577), and relative_military_strength
	# is the army. Legitimacy and gold are in neither term.
	#
	# So this now does what vanilla's own civil-war disasters do. Measured
	# across all 32 vanilla disasters: the tool of the trade is REBELS -
	# monthly_rebel_growth appears in 9 of them - not a '-20% army' sticker,
	# which appears in none. Rebels tie down and destroy the army, and the
	# army is what the gate weighs. The four values below are
	# english_civil_war.txt:47-50 exactly.
	modifier = {
		global_max_control = -0.1
		global_monthly_control = -0.0005
		monthly_legitimacy = -0.03
		nobles_estate_levy_size = -0.25
		pop_join_rebel_threshold = 0.05
		monthly_rebel_growth = 0.01
		monthly_war_exhaustion = 0.15
		global_manpower_modifier = -0.33
	}

	on_start = {
		# THE ALLIANCES GO FIRST, and this is the change that matters.
		#
		# defensive_alliance_strength sums the nation AND everyone who can be
		# called in to defend it, so a Bohemia with two solid allies fails the
		# Ascension's gate for the whole game no matter how badly its own
		# realm is doing. Nothing else in this disaster could reach that term.
		#
		# It is also the history. In 1618 the estates deposed their king and
		# the Protestant Union declined to fight for the man they elected in
		# his place; in 1741 half the Bohemian nobility swore to a rival
		# claimant. Nobody stays bound to a crown that may not be a crown next
		# year. This names no one and benefits no one in particular - every
		# neighbour gains the same opening, and a player holding Bohemia pays
		# the same price.
		#
		# Pattern: generic_actions/general_religion.txt:195-205.
		save_scope_as = PD_boh_crown
		every_related_country = {
			type = alliance
			remove_relation = {
				type = relation_type:alliance
				first = this
				second = scope:PD_boh_crown
			}
		}
		# Two counters, not one, so that both answers are real: buying the
		# estates off ends this with the crown weakened and the realm intact,
		# crushing them ends it with the realm cowed and the treasury spent.
		# Neither is the correct answer. Pattern from turmoil_in_brandenburg,
		# whose end trigger reads two counters the same way
		# (disaster_triggers.txt:707-716).
		set_variable = {
			name = PD_boh_concessions
			value = 0
		}
		set_variable = {
			name = PD_boh_repression
			value = 0
		}
		set_variable = {
			name = PD_boh_crisis_months
			value = 0
		}
		# Which crisis, by DATE rather than by game rule, so the text always
		# suits the century it fires in. Under the windows above this is the
		# revolt in frontloaded and the foreign claimant in strict, but it
		# stays right if a campaign runs long or short.
		if = {
			limit = {
				current_date < 1650.1.1
			}
			trigger_event_non_silently = pd_bohemia.1
		}
		else = {
			trigger_event_non_silently = pd_bohemia.2
		}
		# The neighbours hear of it. No mention of who benefits.
		every_country = {
			limit = {
				this != root
				# All of Europe, not just the German neighbours: a kingdom that
				# cannot govern itself is continental news, and hearing it from
				# everywhere is part of what stops this reading as a private
				# message to Prussia.
				has_presence_in = continent:europe
			}
			trigger_event_non_silently = pd_bohemia.20
		}
	}

	on_monthly = {
		change_variable = {
			name = PD_boh_crisis_months
			add = 1
		}
		# The estates come back to the table every two years until one side
		# has had enough.
		#
		# has_variable first: on_monthly is walked when the disaster's TOOLTIP
		# is drawn, where no country is bound and the counter does not exist.
		# See the note on the end trigger in PD_scripted_triggers.txt.
		if = {
			limit = {
				has_variable = PD_boh_crisis_months
				var:PD_boh_crisis_months >= 24
			}
			set_variable = {
				name = PD_boh_crisis_months
				value = 0
			}
			trigger_event_non_silently = pd_bohemia.10
		}
	}

	on_end = {
		# Guarded for the same reason as on_monthly above.
		if = {
			limit = {
				has_variable = PD_boh_concessions
				var:PD_boh_concessions >= 100
			}
			trigger_event_non_silently = pd_bohemia.100
		}
		else_if = {
			limit = {
				has_variable = PD_boh_repression
				var:PD_boh_repression >= 100
			}
			trigger_event_non_silently = pd_bohemia.101
		}
		remove_variable = PD_boh_concessions
		remove_variable = PD_boh_repression
		remove_variable = PD_boh_crisis_months
		remove_variable = PD_boh_estates_aggrieved
	}
}
